# render.yaml - Blueprint per il deploy su Render.
#
# UN SOLO servizio: server.py serve sia il protocollo WebSocket
# (garage + arena) sia le pagine garage.html/arena.html dallo stesso
# processo/porta (stesso pattern gia' usato da Pac-Man Arena, vedi
# health_check/serve_client). Niente static site separato: la pagina
# richiesta con un GET normale arriva direttamente dal Web Service.
#
# Uso: da dashboard Render -> "New +" -> "Blueprint", punta al repo
# che contiene questo file.
#
# NOTA se stai AGGIORNANDO un servizio Render gia' esistente (es. era
# quello di Pac-Man Arena): il Blueprint NON tocca da solo un servizio
# creato manualmente in passato. Vai su quel servizio -> Settings e
# aggiorna a mano:
#   Start Command: python3 main.py
#   Build Command: pip install -r requirements.txt
# altrimenti continuera' a girare il vecchio main.py.

services:
  - type: web
    name: robot-arena-backend
    runtime: python
    plan: free
    buildCommand: pip install -r requirements.txt
    startCommand: python3 main.py
    envVars:
      - key: DB_PATH
        value: /var/data/robot_arena.db
    disk:
      name: robot-arena-data
      mountPath: /var/data
      sizeGB: 1
